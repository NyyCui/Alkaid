set(PACKAGE_DEPS "${CMAKE_SOURCE_DIR}/third_party/package")
file(READ "${CMAKE_SOURCE_DIR}/package-dependencies.lock.json" PACKAGE_LOCK)
string(JSON PACKAGE_FILE_COUNT LENGTH "${PACKAGE_LOCK}" files)
math(EXPR PACKAGE_LAST_FILE "${PACKAGE_FILE_COUNT} - 1")
foreach(INDEX RANGE ${PACKAGE_LAST_FILE})
  string(JSON FILE GET "${PACKAGE_LOCK}" files ${INDEX} path)
  string(JSON HASH GET "${PACKAGE_LOCK}" files ${INDEX} sha256)
  if(NOT EXISTS "${PACKAGE_DEPS}/${FILE}")
    message(FATAL_ERROR "缺少包管理依赖，请运行 python scripts/prepare-package-deps.py")
  endif()
  file(SHA256 "${PACKAGE_DEPS}/${FILE}" ACTUAL_HASH)
  if(NOT ACTUAL_HASH STREQUAL HASH)
    message(FATAL_ERROR "包管理依赖哈希不匹配：${FILE}")
  endif()
endforeach()
foreach(FILE miniz/miniz.c sqlite/sqlite3.c json/nlohmann/json.hpp)
  if(NOT EXISTS "${PACKAGE_DEPS}/${FILE}")
    message(FATAL_ERROR "缺少锁定的包管理依赖：${FILE}，请先准备 package-dependencies.lock.json 中的文件")
  endif()
endforeach()
add_library(alkaid_package_deps STATIC
  "${PACKAGE_DEPS}/miniz/miniz.c" "${PACKAGE_DEPS}/miniz/miniz_tinfl.c"
  "${PACKAGE_DEPS}/miniz/miniz_tdef.c" "${PACKAGE_DEPS}/miniz/miniz_zip.c" "${PACKAGE_DEPS}/sqlite/sqlite3.c")
target_include_directories(alkaid_package_deps PUBLIC "${PACKAGE_DEPS}/miniz" "${PACKAGE_DEPS}/sqlite")
target_compile_definitions(alkaid_package_deps PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION _CRT_SECURE_NO_WARNINGS)
target_compile_options(alkaid_package_deps PRIVATE /Z7)
add_library(alkaid_packages STATIC package_verifier.cpp package_store.cpp)
target_include_directories(alkaid_packages PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${PACKAGE_DEPS}/json")
target_compile_definitions(alkaid_packages PUBLIC JSON_NOEXCEPTION NOMINMAX _HAS_EXCEPTIONS=0)
target_compile_options(alkaid_packages PRIVATE /W4 /utf-8 /Z7)
target_link_libraries(alkaid_packages PUBLIC alkaid_runtime alkaid_package_deps bcrypt)
add_executable(alkaid_package_tool ../../tools/package_tool.cpp)
target_compile_options(alkaid_package_tool PRIVATE /utf-8 /Z7)
target_link_libraries(alkaid_package_tool PRIVATE alkaid_packages)
add_executable(alkaid_serial_worker ../../plugins/serial/main.cpp)
target_compile_options(alkaid_serial_worker PRIVATE /utf-8 /Z7)
target_link_libraries(alkaid_serial_worker PRIVATE alkaid_packages)

  RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/serial-plugin")
