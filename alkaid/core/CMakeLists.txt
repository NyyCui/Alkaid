# 运行核心不依赖 CEF、Windows API 或前端框架。
add_library(alkaid_runtime STATIC runtime/session_registry.cpp rpc/call_gateway.cpp)
target_include_directories(alkaid_runtime PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(alkaid_runtime PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(alkaid_runtime PRIVATE /W4 /WX /utf-8 /permissive-)
else()
  target_compile_options(alkaid_runtime PRIVATE -Wall -Wextra -Werror)
endif()
if(BUILD_TESTING)
  add_executable(alkaid_runtime_tests ../tests/core/runtime_tests.cpp)
  target_link_libraries(alkaid_runtime_tests PRIVATE alkaid_runtime)
  if(MSVC)
    target_compile_options(alkaid_runtime_tests PRIVATE /W4 /WX /utf-8 /permissive-)
  endif()
  add_test(NAME runtime_boundaries COMMAND alkaid_runtime_tests)
endif()
